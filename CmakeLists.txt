cmake_minimum_required(VERSION 3.20)

project(RayTracer)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include_directories(
    include
    third_party
)

find_package(SDL3 REQUIRED)

add_executable(RayTracer
    src/main.cpp
    src/vec3.cpp
    src/ray.cpp
    src/camera.cpp
    src/window.cpp
    src/framebuffer.cpp
    src/material.cpp
    src/light.cpp
    src/sphere.cpp
    src/plane.cpp
    src/scene.cpp
    src/aabb.cpp
    src/renderer.cpp
    src/bvhNode.cpp
        include/threadPool.h
        src/threadPool.cpp
)

target_link_libraries(RayTracer PRIVATE SDL3::SDL3)

target_compile_options(RayTracer PRIVATE
    -O3
    -march=native
    -Wall
    -Wextra
    -Wpedantic
)
